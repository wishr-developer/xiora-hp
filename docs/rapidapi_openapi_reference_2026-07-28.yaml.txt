info:
  title: xiora API — JP Business Data & Text Utilities
  version: "1.0.0"
  summary: 3 xiora core APIs published on RapidAPI Hub.
  description: |
    xiora が 自社 SaaS (Nexa / XCloud Connect / Xiora Lingua) の 内部 で
    実 使用 して いる 3 API を 外部 提供 する。 全 endpoint は Tokyo VPS
    (xiora-official.com、 Caddy reverse proxy) で 稼働。

    ## Auth
    - RapidAPI Hub 経由 で 契約、 `X-RapidAPI-Key` ヘッダ 必須。
    - 直接 呼出 (RapidAPI 経由 でない) は 401 で reject。

    ## Rate limit tiers (Provider dashboard で 制御)
    - Freemium: 100 req/day
    - Basic (¥1,980/月): 10,000 req/月 (~330 req/day)
    - Pro (¥9,800/月): 100,000 req/月 (~3,300 req/day)
    - Enterprise (¥49,800/月): 1,000,000 req/月

    ## 除外 領域
    投資 助言 · 士業 独占 業務 · 医療 診断 · 薬機 法 該当 情報 は 一切 含めない。

    ## Fair use
    全 endpoint は 出典 明示 · 公開 情報 wrapper。 政府 公開 情報 (国税庁 法人 番号 API)
    は そのまま 転載 する 場合 の 商用 利用 制限 は 無し (国税庁 利用 規約 準拠)。

paths:
  /v1/companies/{corporate_number}:
    get:
      tags: [business-registry]
      operationId: getCompanyByCorporateNumber
      summary: 法人 番号 で 会社 情報 を lookup
      description: |
        国税庁 法人 番号 API を 30 日 cache 付き で wrapper。
        13 桁 の 法人 番号 で 会社 名 · 所在 地 · 設立 日 を 返却。
      parameters:
        - name: corporate_number
          in: path
          required: true
          schema:
            type: string
            pattern: "^[0-9]{13}$"
            example: "2011101044774"
      responses:
        "200":
          description: 会社 情報
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/Company"
              example:
                corporate_number: "2011101044774"
                name: "株式会社 example"
                name_kana: "カブシキガイシャ エグザンプル"
                address: "東京都渋谷区道玄坂1-10-8"
                postal_code: "150-0043"
                established_date: "2020-04-01"
                status: "active"
                source: "kokuzei-hojin-bangou-api"
                cached_at: "2026-07-28T14:00:00+09:00"
        "400":
          $ref: "#/components/responses/BadRequest"
        "401":
          $ref: "#/components/responses/Unauthorized"
        "404":
          description: 該当 法人 番号 なし
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/Error"
        "429":
          $ref: "#/components/responses/RateLimited"

  /v1/extract/contact:
    post:
      tags: [contact-extract]
      operationId: extractContact
      summary: HP から 会社 情報 抽出 (email · phone · 住所 · 代表 名)
      description: |
        URL または HTML string を 入力、 会社 の contact 情報 を LLM 補助 で 抽出。
        confidence_scores で 各 field の 抽出 信頼 度 を 提供。
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              properties:
                url:
                  type: string
                  format: uri
                  example: "https://xiora-official.com/company.html"
                html:
                  type: string
                  maxLength: 500000
              oneOf:
                - required: [url]
                - required: [html]
      responses:
        "200":
          description: 抽出 結果
          content:
            application/json:
              schema:
                type: object
                properties:
                  source:
                    type: string
                  email:
                    type: array
                    items:
                      type: string
                      format: email
                  phone:
                    type: array
                    items:
                      type: string
                  address:
                    type: string
                    nullable: true
                  representative_name:
                    type: string
                    nullable: true
                  company_name:
                    type: string
                    nullable: true
                  confidence_scores:
                    type: object
                    properties:
                      email:
                        type: number
                      phone:
                        type: number
                      address:
                        type: number
                      representative_name:
                        type: number
                      company_name:
                        type: number
              example:
                source: "https://xiora-official.com/company.html"
                email: ["info@xiora-official.com"]
                phone: ["070-9165-0203"]
                address: "〒150-0043 東京都渋谷区道玄坂1-10-8"
                representative_name: "沓澤 怜士"
                company_name: "xiora"
                confidence_scores:
                  email: 0.98
                  phone: 0.95
                  address: 0.91
                  representative_name: 0.82
                  company_name: 0.97
        "400":
          $ref: "#/components/responses/BadRequest"
        "401":
          $ref: "#/components/responses/Unauthorized"
        "429":
          $ref: "#/components/responses/RateLimited"
